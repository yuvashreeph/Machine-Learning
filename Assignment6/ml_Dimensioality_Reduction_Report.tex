\section{Data Preprocessing}
\subsection{Data Loading and Cleaning}
\begin{lstlisting}[language=Python, caption={Loading, cleaning, and preprocessing data}]
import pandas as pd
import numpy as np
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import train_test_split

# Load data
df = pd.read_csv('spambase_csv.csv')
print("Initial DataFrame head:")
print(df.head())
print("\nInitial shape:", df.shape)

# Impute missing values with the median
df.fillna(df.median(numeric_only=True), inplace=True)

# Remove outliers using the Z-score method (threshold=3)
def remove_outliers(df, threshold=3):
    numeric_cols = df.select_dtypes(include=[np.number]).columns
    z_scores = np.abs((df[numeric_cols] - df[numeric_cols].mean()) / df[numeric_cols].std())
    return df[(z_scores < threshold).all(axis=1)]

df_cleaned = remove_outliers(df)
print("\nShape after cleaning:", df_cleaned.shape)

# Feature Scaling and Data Splitting
X = df_cleaned.drop(columns=['class'])
y = df_cleaned['class']
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)
X_train, X_test, y_train, y_test = train_test_split(
    X_scaled, y, test_size=0.2, stratify=y, random_state=42
)
\end{lstlisting}

\section{Conclusion}
\begin{itemize}
    \item \textbf{Performance Hierarchy:} A clear performance hierarchy was observed. Simple probabilistic models like Naive Bayes performed the poorest, while complex ensemble methods performed the best. The high training accuracy on tree-based models suggests some overfitting, which could be addressed with further hyperparameter tuning.
    \item \textbf{Superiority of Ensemble Methods:} Tree-based ensemble models significantly outperformed all other individual classifiers, highlighting their ability to capture complex patterns in the data.
    \item \textbf{XGBoost as the Champion Model:} The \textbf{XGBoost Classifier} was unequivocally the best-performing model, achieving the highest test accuracy of \textbf{94.00\%}. This demonstrates its robustness and state-of-the-art performance for this classification task.
\end{itemize}

\section{Learning Outcomes}
\begin{itemize}
    \item Gained hands-on experience performing hyperparameter tuning for multiple classifiers including SVM, KNN, Naive Bayes, Logistic Regression, Decision Tree, Random Forest, AdaBoost, Gradient Boosting, and XGBoost.
    \item Learned to apply PCA for dimensionality reduction and observed its impact on model performance, variance, and overfitting.
    \item Understood how different models respond to PCA, identifying which models benefit most (Naive Bayes, KNN) and which are robust to high-dimensional data (ensemble methods, linear models).
    \item Observed how stacking ensembles can improve predictive performance and maintain robustness to dimensionality reduction compared to individual models.
\end{itemize}
